cmake_minimum_required(VERSION 3.24)
# Use the isolated compiler wheel on Linux; no machine-wide toolchain install.
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux")
  execute_process(COMMAND "${Python_EXECUTABLE}" -c "import pathlib,ziglang; print(pathlib.Path(ziglang.__file__).parent/'zig')"
                  OUTPUT_VARIABLE ZIG OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cc" "#!/bin/sh\nexec '${ZIG}' cc \"$@\"\n")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cxx" "#!/bin/sh\nexec '${ZIG}' c++ \"$@\"\n")
  file(CHMOD "${CMAKE_CURRENT_BINARY_DIR}/cc" "${CMAKE_CURRENT_BINARY_DIR}/cxx"
       PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  set(CMAKE_C_COMPILER "${CMAKE_CURRENT_BINARY_DIR}/cc")
  set(CMAKE_CXX_COMPILER "${CMAKE_CURRENT_BINARY_DIR}/cxx")
endif()
project(myvoice_aec3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/pywebrtc-audio-0.2.0.tar.gz")
file(SHA256 "${ARCHIVE}" SOURCE_SHA256)
if(NOT SOURCE_SHA256 STREQUAL "be04c2815fed409c3cf64ae70ca267208c3067a155794dd2c09e6e114899ec1f")
  message(FATAL_ERROR "Vendored WebRTC source checksum mismatch")
endif()
file(ARCHIVE_EXTRACT INPUT "${ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/upstream")
set(UPSTREAM "${CMAKE_CURRENT_BINARY_DIR}/upstream/pywebrtc-audio-0.2.0")
add_subdirectory("${UPSTREAM}/vendor/webrtc_audio" webrtc_audio)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(myvoice_aec3 aec3.cpp)
target_link_libraries(myvoice_aec3 PRIVATE webrtc_audio_static)
target_include_directories(myvoice_aec3 PRIVATE "${UPSTREAM}/vendor/webrtc_audio")
install(TARGETS myvoice_aec3 DESTINATION .)
install(FILES "${UPSTREAM}/LICENSE" "${UPSTREAM}/NOTICE" DESTINATION myvoice_aec3_licenses)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/WEBRTC-LICENSE" DESTINATION myvoice_aec3_licenses)
install(DIRECTORY "${UPSTREAM}/vendor/webrtc_audio/" DESTINATION myvoice_aec3_licenses/webrtc
        FILES_MATCHING PATTERN "*LICENSE*" PATTERN "*COPYING*" PATTERN "PATENTS" PATTERN "AUTHORS")
